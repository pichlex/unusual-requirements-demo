import { HapticRange } from './haptics.js';

const slider = new HapticRange({
  surface,
  driver,
  step: 10,
  isMajor: value => value % 20 === 0,
  accentOffset: 1.35,
  onTick: ({ value, kind }) => {
    animateTick(value, kind);
  },
});

// No synthetic click(). The input remains the real touch target.
